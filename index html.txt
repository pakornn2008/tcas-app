        // File Handler: Convert File to Base64
        function readFileAsBase64(file) {
            return new Promise((resolve) => {
                if (!file) return resolve(null);
                const reader = new FileReader();
                reader.onload = e => resolve({
                    name: file.name,
                    mimeType: file.type,
                    base64: e.target.result
                });
                reader.readAsDataURL(file);
            });
        }

        // Lightbox Preview Handler
        function openImagePreview(url, title = 'ไฟล์หลักฐาน', mimeType = '') {
    const modal = document.getElementById('imagePreviewModal');
    const img = document.getElementById('previewModalImage');
    const iframe = document.getElementById('previewModalIframe');
    document.getElementById('previewModalTitle').innerText = title;

    // 1. เคลียร์ event onerror เก่าเพื่อไม่ให้ตกค้างจากการเปิดครั้งก่อน
    img.onerror = null;
    iframe.onerror = null;

    // 2. ซ่อน element แสดงผลไว้ก่อน
    img.classList.add('hidden');
    iframe.classList.add('hidden');

    // 3. ฟังก์ชันการทำงานเมื่อโหลดไฟล์ไม่สำเร็จ
    const handleFallback = () => {
        modal.classList.add('hidden'); // ปิด Modal
        window.open(url, '_blank');   // เปิดแท็บใหม่
    };

    const isPdf = mimeType.includes('pdf') || url.toLowerCase().includes('.pdf');

    if (isPdf) {
        iframe.onerror = handleFallback;
        iframe.src = url;
        iframe.classList.remove('hidden');
    } else {
        img.onerror = handleFallback;
        img.src = url;
        img.classList.remove('hidden');
    }

    modal.classList.remove('hidden');
}

        // Handle Admin Search & Filtering (รองรับการค้นหาตาม เลขประจำตัวนักเรียน)
        function handleAdminSearchAndFilter() {
            const searchKeyword = (document.getElementById('adminSearchInput').value || '').toLowerCase().trim();
            const statusFilter = document.getElementById('adminStatusFilter').value;
            const catFilter = document.getElementById('adminCatFilter').value;
            const gradeFilter = document.getElementById('adminGradeFilter').value;

            filteredAdminList = adminDataList.filter(item => {
                const fullName = `${item.title || ''}${item.firstName || ''} ${item.lastName || ''}`.toLowerCase();
                const cid = String(item.citizenId || '').toLowerCase();
                const stId = String(item.studentId || '').toLowerCase();
                const pTitle = String(item.programTitle || '').toLowerCase();

                const matchesKeyword = !searchKeyword || fullName.includes(searchKeyword) || cid.includes(searchKeyword) || stId.includes(searchKeyword) || pTitle.includes(searchKeyword);
                const matchesStatus = (statusFilter === 'ALL') || (item.status === statusFilter);
                const matchesCat = (catFilter === 'ALL') || (item.tcasCategory === catFilter);
                const matchesGrade = (gradeFilter === 'ALL') || (item.grade === gradeFilter);

                return matchesKeyword && matchesStatus && matchesCat && matchesGrade;
            });

            currentPage = 1;
            renderAdminTable();
        }

        function filterByStatus(status) {
            document.getElementById('adminStatusFilter').value = status;
            handleAdminSearchAndFilter();
        }

        // Row Selection Logic for Bulk Operations
        function toggleRowSelection(id, isSelected) {
            if (isSelected) {
                selectedRowIds.add(String(id));
            } else {
                selectedRowIds.delete(String(id));
            }
            updateBulkActionBar();
        }

        function toggleSelectAllRows(isChecked) {
            if (isChecked) {
                filteredAdminList.forEach(item => selectedRowIds.add(String(item.id)));
            } else {
                selectedRowIds.clear();
            }
            renderAdminTable();
            updateBulkActionBar();
        }

        function updateBulkActionBar() {
            const container = document.getElementById('bulkActionsContainer');
            const countText = document.getElementById('selectedCountText');
            if (selectedRowIds.size > 0) {
                container.classList.remove('hidden');
                countText.innerText = `เลือก ${selectedRowIds.size} รายการ`;
            } else {
                container.classList.add('hidden');
            }
        }

        function bulkApproveSelected() {
            if (confirm(`คุณต้องการอนุมัติผลงานที่เลือกทั้ง ${selectedRowIds.size} รายการใช่หรือไม่?`)) {

                fetchAdminData();
                showToast('อนุมัติรายการที่เลือกเรียบร้อยแล้ว', 'success');
            }
        }

                selectedRowIds.forEach(id => deleteItem(id, false));

                updateBulkActionBar();
                fetchAdminData();
                showToast('ลบรายการที่เลือกเรียบร้อยแล้ว', 'info');

        }

            const remarkText = document.getElementById('adminRemarkInput').value.trim();

        // Update Item Status & Remark in Google Sheet or LocalStorage
        function updateStatusAndRemark(id, newStatus, remarkText, refresh = true) {
            if (typeof google !== 'undefined' && google.script && google.script.run) {
                google.script.run
                    .withSuccessHandler((res) => {
                        if (res.success && refresh) {
                            showToast(`อัปเดตสถานะเป็น ${STATUS_TEXT[newStatus]} เรียบร้อยแล้ว`, 'success');
                            fetchAdminData();
                        }
                    })
                    .updateStatusInSheet(id, newStatus, remarkText);
            } else {
                const localData = getLocalData();
                const idx = localData.findIndex(item => String(item.id) === String(id));
                if (idx !== -1) {
                    localData[idx].status = newStatus;
                    localData[idx].remark = remarkText;
                    saveLocalData(localData);
                    if (refresh) {
                        fetchAdminData();
                        showToast(`อัปเดตสถานะเป็น ${STATUS_TEXT[newStatus]} เรียบร้อยแล้ว`, 'success');
                    }
                }
            }
        }

        // Delete Item
        function deleteItem(id, refresh = true) {
            if (!refresh || confirm('คุณต้องการลบข้อมูลผลงานนี้ออกจากระบบใช่หรือไม่?')) {
                if (typeof google !== 'undefined' && google.script && google.script.run) {
                    google.script.run
                        .withSuccessHandler((res) => {
                            if (res.success && refresh) {
                                showToast('ลบข้อมูลออกจากระบบแล้ว', 'info');
                                fetchAdminData();
                            }
                        })
                        .deleteActivityFromSheet(id);
                } else {
                    const localData = getLocalData().filter(item => String(item.id) !== String(id));
                    saveLocalData(localData);
                    if (refresh) {
                        fetchAdminData();
                        showToast('ลบข้อมูลออกจากระบบแล้ว', 'info');
                    }
                }
            }
        }

        // Export Data to CSV (รวมฟิลด์ เลขประจำตัวนักเรียน)
        function exportToCSV() {
    if (!adminDataList || adminDataList.length === 0) {
        return showToast('ไม่มีข้อมูลสำหรับส่งออก', 'warning');
    }

    // 1. กรองดึงมาเฉพาะรายการที่สถานะเป็น 'approve' หรือ 'approved' หรือ 'อนุมัติ'
    const approvedData = adminDataList.filter(item => {
        const status = (item.status || '').toLowerCase().trim();
        return status === 'approve' || status === 'approved' || status.includes('อนุมัติ');
    });

    if (approvedData.length === 0) {
        return showToast('ไม่มีข้อมูลที่ได้รับการอนุมัติ (Approved) สำหรับส่งออก', 'warning');
    }

    // 2. แยกข้อมูลตามประเภทแบบฟอร์ม (formType)
    const activities = approvedData.filter(item => (item.formType || '').toLowerCase() === 'activity');
    const prizes = approvedData.filter(item => (item.formType || '').toLowerCase() === 'prizes' || (item.formType || '').toLowerCase() === 'prize');
    const projects = approvedData.filter(item => (item.formType || '').toLowerCase() === 'project');

    let exportedCount = 0;
    const dateStr = new Date().toISOString().slice(0, 10);

    // ฟังก์ชันย่อยสำหรับสร้างและดาวน์โหลดไฟล์ CSV ตามโครงสร้างที่กำหนด
    const downloadFile = (filename, headers, dataArray, mapFn) => {
        if (dataArray.length === 0) return; // ถ้าไม่มีข้อมูลในหมวดนี้ ไม่ต้องโหลดไฟล์

        let csvContent = "\uFEFF"; // UTF-8 BOM ทำให้เปิดใน Excel แล้วภาษาไทยไม่เพี้ยน
        csvContent += headers.join(",") + "\n";

        dataArray.forEach(item => {
            const row = mapFn(item);
            csvContent += row.join(",") + "\n";
        });

        const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
        const link = document.createElement("a");
        const url = URL.createObjectURL(blob);
        link.setAttribute("href", url);
        link.setAttribute("download", filename);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(url);
        
        exportedCount++;
    };

    // ---------------------------------------------------------
    // 3. ส่งออกไฟล์ 01-activities
    // ---------------------------------------------------------
    downloadFile(
        `01-activities_${dateStr}.csv`,
