/**

    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#fdf2f8').setFontColor('#be185d');

    sheet.setColumnWidth(22, 250); // ไฟล์รูปกิจกรรม
  }
  return sheet;
}

/**

  }
}

/**
 * จัดการสร้างโฟลเดอร์ย่อยใน Google Drive (Certificates และ Photos)
 */
function setupDriveSubfolders() {
  try {
    const parentFolder = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID);
    let certFolder, photoFolder;
    
    // ค้นหาโฟลเดอร์ย่อย
    const folders = parentFolder.getFolders();
    while (folders.hasNext()) {
      const folder = folders.next();
      if (folder.getName() === 'Certificates') certFolder = folder;
      if (folder.getName() === 'Photos') photoFolder = folder;
    }
    
    // สร้างถ้ายังไม่มี
    if (!certFolder) certFolder = parentFolder.createFolder('Certificates');
    if (!photoFolder) photoFolder = parentFolder.createFolder('Photos');
    
    return { certFolder, photoFolder };
  } catch (error) {
    logEvent('ERROR', 'setupDriveSubfolders', error.toString());
    throw new Error('ไม่สามารถเข้าถึงหรือสร้างโฟลเดอร์ใน Google Drive ได้');
  }
}

/**
 * แปลง Base64 และบันทึกไฟล์ลง Google Drive พร้อมสร้าง URL สำหรับใช้งาน
 */
function saveFileToDrive(fileObj, prefix, targetFolder) {
  if (!fileObj || !fileObj.base64) return null;
  
  try {
    let mimeType = 'application/octet-stream';
    let base64Data = fileObj.base64;
    
    if (base64Data.indexOf('base64,') !== -1) {
      const parts = base64Data.split('base64,');
      base64Data = parts[1];
      const meta = parts[0];
      if (meta.indexOf('data:') !== -1 && meta.indexOf(';') !== -1) {
        mimeType = meta.substring(meta.indexOf('data:') + 5, meta.indexOf(';'));
      }
    }
    
    const decodedData = Utilities.base64Decode(base64Data);
    
    let extension = '';
    if (mimeType.includes('pdf')) extension = '.pdf';
    else if (mimeType.includes('jpeg') || mimeType.includes('jpg')) extension = '.jpg';
    else if (mimeType.includes('png')) extension = '.png';
    
    const safeName = (fileObj.name || 'file').replace(/[^\w\s.-]/gi, '');
    const fileName = `${prefix}_${Date.now()}_${safeName}${!safeName.includes('.') ? extension : ''}`;
    const blob = Utilities.newBlob(decodedData, mimeType, fileName);
    
    const driveFile = targetFolder.createFile(blob);
    const fileId = driveFile.getId(); 
    
    try {
      driveFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (e) {
      logEvent('WARNING', 'File Sharing Restriction', e.toString(), fileId);
    }
    
    return {
      id: fileId,
      name: fileName,
      mimeType: mimeType,
      size: fileObj.size || 0,
      url: driveFile.getUrl(),
      previewUrl: `https://drive.google.com/file/d/${fileId}/preview`,
      viewUrl: `https://drive.google.com/uc?export=view&id=${fileId}`,
      downloadUrl: `https://drive.google.com/uc?export=download&id=${fileId}`,
      thumbnailUrl: `https://drive.google.com/thumbnail?id=${fileId}&sz=w1000`
    };
  } catch (err) {
    logEvent('ERROR', 'saveFileToDrive', err.toString());
    throw new Error(`การอัปโหลดไฟล์ ${prefix} ล้มเหลว: ${err.message}`);
  }
}

/**
 * ฟังก์ชันตรวจสอบความถูกต้องของข้อมูลพื้นฐานก่อนบันทึก
 */
function validateFormData(formData) {
  if (!formData.firstName || !formData.lastName) return "กรุณากรอกชื่อและนามสกุล";
  if (!formData.citizenId || formData.citizenId.length !== 13) return "เลขบัตรประชาชนต้องมี 13 หลัก";
  if (!formData.studentId || formData.studentId.length !== 5) return "เลขประจำตัวนักเรียนต้องมี 5 หลัก";
  if (!formData.programTitle) return "กรุณาระบุชื่อโครงการ/กิจกรรม";
  return null;
}

/**
 * ฟังก์ชันรับข้อมูลจาก Frontend และบันทึกลงชีต
 */
function saveActivityToDrive(formData) {
  try {
    const validationError = validateFormData(formData);
    if (validationError) {
      return { success: false, message: validationError };
    }

    const folders = setupDriveSubfolders();
    const sheet = getSheet();
    const newId = formData.id || `ACT-${Date.now()}`;
    
    let certFileData = null;
    let photoFileData = null;
    
    if (formData.certFile && formData.certFile.base64) {
      certFileData = saveFileToDrive(formData.certFile, `CERT_${formData.studentId}`, folders.certFolder);
    }
    if (formData.photoFile && formData.photoFile.base64) {
      photoFileData = saveFileToDrive(formData.photoFile, `PHOTO_${formData.studentId}`, folders.photoFolder);
    }
    
    const certJsonStr = certFileData ? JSON.stringify(certFileData) : '';
    const photoJsonStr = photoFileData ? JSON.stringify(photoFileData) : '';
    
    // จัดเรียงข้อมูล 26 คอลัมน์
    const rowData = [
      newId,
      formData.title || '',
      formData.firstName || '',
      formData.lastName || '',
      `'${formData.studentId || ''}`, 
      `'${formData.citizenId || ''}`,
      formData.grade || '',
      formData.room || '',
      formData.programTrack || '',
      formData.formType || '',
      formData.tcasCategory || '',
      formData.programTitle || '',
      formData.dynamicField || '',
      formData.description || '',
      formData.startDate || '',
      formData.endDate || '',
      formData.year || new Date().getFullYear() + 543,
      formData.hours || 0, // เพิ่ม hours
      formData.fee || 0,   // เพิ่ม fee
      formData.level || '',
      certJsonStr,
      photoJsonStr,
      formData.newsUrl || '',
      formData.status || 'PENDING',
      formData.remark || '',
      new Date().toLocaleString('th-TH')
    ];
    
    sheet.appendRow(rowData);
    logEvent('INFO', 'Create Activity', `Saved new activity for ${formData.studentId}`, newId);
    
    return {
      success: true,
      message: 'บันทึกข้อมูลและอัปโหลดหลักฐานสำเร็จเรียบร้อยแล้ว'
    };
  } catch (error) {
    logEvent('ERROR', 'saveActivityToDrive', error.toString(), formData.studentId);
    return {
      success: false,
      message: `เกิดข้อผิดพลาดในการบันทึกข้อมูล: ${error.message}`
    };
  }
}

/**
 * ดึงข้อมูลกิจกรรมทั้งหมด (สำหรับฝั่ง Admin)
 */
function getAllActivities() {
  try {
    const sheet = getSheet();
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return []; 
    
    // อ่านข้อมูลทั้งหมด 26 คอลัมน์
    const data = sheet.getRange(2, 1, lastRow - 1, 26).getDisplayValues();
    const result = [];
    
    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      let certFileObj = null;
      let photoFileObj = null;
      
      if (row[20]) {
        try { certFileObj = JSON.parse(row[20]); } catch (e) { console.warn("Failed to parse cert json at row " + (i+2)); }
      }
      if (row[21]) {
        try { photoFileObj = JSON.parse(row[21]); } catch (e) { console.warn("Failed to parse photo json at row " + (i+2)); }
      }
      
      result.push({
        id: String(row[0]),
        title: String(row[1]),
        firstName: String(row[2]),
        lastName: String(row[3]),
        studentId: String(row[4]).replace(/'/g, ""),
        citizenId: String(row[5]).replace(/'/g, ""),
        grade: String(row[6]),
        room: String(row[7]),
        programTrack: String(row[8]),
        formType: String(row[9]),
        tcasCategory: String(row[10]),
        programTitle: String(row[11]),
        dynamicField: String(row[12]),
        description: String(row[13]),
        startDate: String(row[14]),
        endDate: String(row[15]),
        year: String(row[16]),
        hours: String(row[17] || 0), // อ่านค่า hours
        fee: String(row[18] || 0),   // อ่านค่า fee
        level: String(row[19]),
        certFile: certFileObj,
        photoFile: photoFileObj,
        newsUrl: String(row[22]),
        status: String(row[23] || 'PENDING'),
        remark: String(row[24]),
        timestamp: String(row[25])
      });
    }
    return result;
  } catch (error) {
    logEvent('ERROR', 'getAllActivities', error.toString());
    return [];
  }
}

/**
 * ฟังก์ชันดึงข้อมูลเฉพาะของนักเรียน
 */
function getStudentActivities(searchId) {
  if (!searchId) return [];
  
  try {
    const allData = getAllActivities();
    const cleanSearchId = String(searchId).trim();
    
    const studentData = allData.filter(item => 
      item.studentId === cleanSearchId || item.citizenId === cleanSearchId
    );
    
    logEvent('INFO', 'Search Student Data', `Found ${studentData.length} records`, cleanSearchId);
    return studentData;
  } catch (error) {
    logEvent('ERROR', 'getStudentActivities', error.toString(), searchId);
    return [];
  }
}

/**
 * อัปเดตสถานะและข้อความแจ้งเตือนแก้ไข (โดย Admin)
 */
function updateStatusInSheet(id, newStatus, remarkText) {
  try {
    if (!id || !newStatus) return { success: false, message: 'ข้อมูลไม่ครบถ้วน' };

    const sheet = getSheet();
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return { success: false, message: 'ไม่พบข้อมูลในฐานข้อมูล' };
    
    const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(id)) {
        const rowIndex = i + 2;
        sheet.getRange(rowIndex, 24).setValue(newStatus);   // คอลัมน์สถานะ (Col 24)
        sheet.getRange(rowIndex, 25).setValue(remarkText || ''); // คอลัมน์ข้อความแจ้งแก้ไข (Col 25)
        
        logEvent('INFO', 'Update Status', `Changed to ${newStatus}, Remark: ${remarkText}`, id);
        return { success: true, message: 'อัปเดตสถานะเรียบร้อยแล้ว' };
      }
    }
    return { success: false, message: 'ไม่พบรายการที่ต้องการอัปเดตรหัสนี้' };
  } catch (error) {
    logEvent('ERROR', 'updateStatusInSheet', error.toString(), id);
    return { success: false, message: 'เกิดข้อผิดพลาดของระบบ: ' + error.message };
  }
}

/**
 * ลบข้อมูลกิจกรรมออกจาก Google Sheet
 */
function deleteActivityFromSheet(id) {
  try {
    if (!id) return { success: false, message: 'ไม่ได้ระบุรหัสกิจกรรมที่ต้องการลบ' };

    const sheet = getSheet();
    const lastRow = sheet.getLastRow();
    if (lastRow <= 1) return { success: false, message: 'ไม่พบข้อมูลในชีต' };
    
    const ids = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]) === String(id)) {
        sheet.deleteRow(i + 2);
        
        logEvent('WARNING', 'Delete Activity', 'Row deleted completely', id);
        return { success: true, message: 'ลบข้อมูลสำเร็จ' };
      }
    }
    return { success: false, message: 'ไม่พบรายการที่ต้องการลบ' };
  } catch (error) {
    logEvent('ERROR', 'deleteActivityFromSheet', error.toString(), id);
    return { success: false, message: 'เกิดข้อผิดพลาดในการลบข้อมูล: ' + error.message };
  }
}
